\textbf{Problem 1} (20 pts). Let $A$ be a $3\times 3$ matrix with eigenvalues $\lambda_1 = 1$, $\lambda_2 = -1$, $\lambda_3 = 2$ and corresponding eigenvectors
\[
v_1 = \begin{pmatrix}1\\0\\1\end{pmatrix},\quad v_2 = \begin{pmatrix}1\\1\\0\end{pmatrix},\quad v_3 = \begin{pmatrix}0\\1\\1\end{pmatrix}.
\]
\begin{enumerate}
    \item[(a)] Diagonalize $A$; that is, write $A = PDP^{-1}$.
    \item[(b)] Compute $A^5$.
\end{enumerate}

\bigskip

\textbf{Problem 2} (20 pts). Find an orthonormal basis for the subspace of $\mathbb{R}^3$ spanned by
\[
u_1 = \begin{pmatrix}1\\1\\0\end{pmatrix},\quad u_2 = \begin{pmatrix}1\\0\\1\end{pmatrix},\quad u_3 = \begin{pmatrix}0\\1\\1\end{pmatrix}
\]
using the Gram--Schmidt process.

\bigskip

\textbf{Problem 3} (20 pts). Consider the dynamical system $x_{t+1} = Ax_t$ where
\[
A = \begin{pmatrix}0.8 & 0.3\\0.2 & 0.7\end{pmatrix}, \quad x_0 = \begin{pmatrix}500\\500\end{pmatrix}.
\]
\begin{enumerate}
    \item[(a)] Find the eigenvalues and eigenvectors of $A$.
    \item[(b)] Find a formula for $x_t$.
    \item[(c)] What is the long-run behavior of $x_t$ as $t \to \infty$?
\end{enumerate}

\bigskip

\textbf{Problem 4} (20 pts). Find the least-squares solution to $Ax = b$ where
\[
A = \begin{pmatrix}1&1\\1&2\\1&3\end{pmatrix}, \quad b = \begin{pmatrix}1\\3\\4\end{pmatrix}.
\]
Interpret this geometrically.

\bigskip

\textbf{Problem 5} (20 pts). Let $V$ be the space of continuous functions on $[0, 2\pi]$. Show that the set $\{1, \cos t, \sin t\}$ is linearly independent in $V$.
