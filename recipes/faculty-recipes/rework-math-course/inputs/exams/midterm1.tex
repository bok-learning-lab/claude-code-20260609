\textbf{Instructions:} You have 90 minutes. No notes, no calculators. Show all work for full credit.

\bigskip

\textbf{Problem 1} (20 pts). Solve the following linear system. Express your answer in parametric form if there are infinitely many solutions.
\[
\begin{cases}
x_1 + 2x_2 - x_3 = 4 \\
2x_1 + 5x_2 + x_3 = 11 \\
x_1 + 3x_2 + 4x_3 = 13
\end{cases}
\]

\bigskip

\textbf{Problem 2} (20 pts). Let $A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$ and $B = \begin{pmatrix} 0 & -1 \\ 1 & 2 \end{pmatrix}$.
\begin{enumerate}
    \item[(a)] Compute $AB$ and $BA$.
    \item[(b)] Is $A$ invertible? If so, find $A^{-1}$.
    \item[(c)] Solve $Ax = \begin{pmatrix}1\\2\end{pmatrix}$.
\end{enumerate}

\bigskip

\textbf{Problem 3} (20 pts). Determine whether the following vectors are linearly independent. Justify your answer.
\[
v_1 = \begin{pmatrix}1\\0\\2\end{pmatrix},\quad v_2 = \begin{pmatrix}0\\1\\-1\end{pmatrix},\quad v_3 = \begin{pmatrix}2\\-1\\5\end{pmatrix}
\]

\bigskip

\textbf{Problem 4} (20 pts). Let $T: \mathbb{R}^2 \to \mathbb{R}^2$ be the linear transformation that first reflects across the $x$-axis and then rotates counterclockwise by $\frac{\pi}{4}$.
\begin{enumerate}
    \item[(a)] Find the standard matrix of $T$.
    \item[(b)] Is $T$ invertible? If so, find $T^{-1}$.
\end{enumerate}

\bigskip

\textbf{Problem 5} (20 pts). Find a basis for each of the following subspaces of $\mathbb{R}^4$, and state the dimension.
\begin{enumerate}
    \item[(a)] The null space of $A = \begin{pmatrix}1&2&0&-1\\0&0&1&2\\1&2&1&1\end{pmatrix}$.
    \item[(b)] The column space of $A$.
\end{enumerate}
