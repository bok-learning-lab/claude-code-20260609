\begin{defn}[Augmented matrix]
The \emph{augmented matrix} of a linear system $A\mathbf{x} = \mathbf{b}$ is the $m \times (n+1)$ matrix $[A \mid \mathbf{b}]$ formed by appending the constant column to the coefficient matrix. Row operations on $[A \mid \mathbf{b}]$ correspond exactly to equation operations on the original system.
\end{defn}

\begin{thm}[Rank-Nullity]
For any $m \times n$ matrix $A$,
\[
\rank(A) + \nullity(A) = n,
\]
where $n$ is the number of columns of $A$.
\end{thm}
